let state = {
  user: null,          // email connecté
  employees: [],       // [{id, name, role}]
  entries: [],         // [{id, empId, date, hours, chantier}]
  recapMonth: null,    // "YYYY-MM"
};

// Couleurs avatar (round-robin)
const AV_COLORS = ['av-0','av-1','av-2','av-3','av-4'];

/* ────── PERSISTANCE ────── */
function save() {
  localStorage.setItem('tt_state', JSON.stringify(state));
}
function load() {
  const raw = localStorage.getItem('tt_state');
  if (raw) {
    try { state = { ...state, ...JSON.parse(raw) }; } catch(e) {}
  }
  if (!state.recapMonth) {
    state.recapMonth = currentYM();
  }
}

/* ────── HELPERS ────── */
function uid() {
  return Date.now().toString(36) + Math.random().toString(36).slice(2);
}
function currentYM() {
  const d = new Date();
  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}`;
}
function ymToLabel(ym) {
  if (!ym) return '—';
  const [y, m] = ym.split('-');
  const months = ['Jan','Fév','Mar','Avr','Mai','Juin','Juil','Août','Sep','Oct','Nov','Déc'];
  return `${months[+m-1]} ${y}`;
}
function initials(name) {
  return name.split(' ').map(p=>p[0]?.toUpperCase()||'').slice(0,2).join('');
}
function getEmployee(id) {
  return state.employees.find(e=>e.id===id);
}
function avClass(idx) { return AV_COLORS[idx % AV_COLORS.length]; }

function toast(msg, color='var(--success)') {
  const el = document.getElementById('toast');
  el.textContent = msg;
  el.style.borderLeftColor = color;
  el.classList.add('show');
  setTimeout(()=>el.classList.remove('show'), 2200);
}

  state.user = null;
  save();

  document.getElementById('screen-login').classList.add('active');

    state.user.length > 18 ? state.user.slice(0,16)+'…' : state.user;
  renderDashboard();
  renderEmployees();

function switchTab(btn, tab) {
  // nav highlight
  document.querySelectorAll('.nav-item').forEach(b=>b.classList.remove('active'));
  btn.classList.add('active');
  // show/hide tabs
  document.querySelectorAll('.tab-content').forEach(t=>t.style.display='none');
  document.getElementById('tab-'+tab).style.display='flex';
  // refresh data
  if (tab==='dashboard') renderDashboard();
  if (tab==='employees') renderEmployees();
  if (tab==='recap')     renderRecap();
  if (tab==='add')       { populateEmpSelect(); setDefaultDate(); }
}

/* ════════════════════════════════════════════════════
   EMPLOYEES
════════════════════════════════════════════════════ */
function addEmployee() {
  const name = document.getElementById('emp-name-input').value.trim();
  const role = document.getElementById('emp-role-input').value.trim();
  if (!name) { toast('⚠ Nom requis.', 'var(--danger)'); return; }
  state.employees.push({ id: uid(), name, role: role||'—' });
  save();
  document.getElementById('emp-name-input').value = '';
  document.getElementById('emp-role-input').value = '';
  closeModal('modal-add-emp');
  renderEmployees();
  populateEmpSelect();
  toast('✓ Salarié ajouté !');
}

function deleteEmployee(id) {
  const emp = getEmployee(id);
  if (!emp) return;
  if (!confirm(`Supprimer ${emp.name} ? Ses saisies seront aussi effacées.`)) return;
  state.employees = state.employees.filter(e=>e.id!==id);
  state.entries   = state.entries.filter(e=>e.empId!==id);
  save();
  renderEmployees();
  populateEmpSelect();
  renderDashboard();
  toast('Salarié supprimé.', 'var(--danger)');
}

function renderEmployees() {
  const list = document.getElementById('emp-list');
  document.getElementById('emp-count').textContent = state.employees.length;
  if (!state.employees.length) {
    list.innerHTML = `<div class="empty"><div class="empty-icon">👤</div><div class="empty-text">Aucun salarié. Appuyez sur "+ Ajouter".</div></div>`;
    return;
  }
  list.innerHTML = state.employees.map((emp, idx) => {
    const total = state.entries.filter(e=>e.empId===emp.id).reduce((s,e)=>s+e.hours,0);
    return `
      <div class="emp-item" style="margin-bottom:10px">
        <div class="emp-avatar ${avClass(idx)}">${initials(emp.name)}</div>
        <div class="emp-info">
          <div class="emp-name">${esc(emp.name)}</div>
          <div class="emp-role">${esc(emp.role)}</div>
        </div>
        <div style="display:flex;align-items:center;gap:8px">
          <div class="emp-hours">${total}h</div>
          <button class="btn btn-icon btn-sm" style="color:var(--danger);border-color:var(--danger)"
            onclick="deleteEmployee('${emp.id}')">✕</button>
        </div>
      </div>`;
  }).join('');
}

/* ════════════════════════════════════════════════════
   ENTRIES
════════════════════════════════════════════════════ */
function setDefaultDate() {
  const d = document.getElementById('entry-date');
  if (!d.value) d.value = new Date().toISOString().split('T')[0];
}

function populateEmpSelect() {
  const sel = document.getElementById('entry-emp');
  const current = sel.value;
  sel.innerHTML = '<option value="">— Choisir un salarié —</option>' +
    state.employees.map(e => `<option value="${e.id}" ${e.id===current?'selected':''}>${esc(e.name)}</option>`).join('');
}

function addEntry() {
  const empId    = document.getElementById('entry-emp').value;
  const date     = document.getElementById('entry-date').value;
  const hours    = parseFloat(document.getElementById('entry-hours').value);
  const chantier = document.getElementById('entry-chantier').value.trim();

  if (!empId)       { toast('⚠ Choisissez un salarié.', 'var(--danger)'); return; }
  if (!date)        { toast('⚠ Date requise.', 'var(--danger)'); return; }
  if (!hours||hours<=0) { toast('⚠ Entrez un nombre d\'heures valide.', 'var(--danger)'); return; }

  state.entries.push({ id: uid(), empId, date, hours, chantier: chantier||'' });
  save();

  // reset form (keep employee & chantier for quick re-entry)
  document.getElementById('entry-hours').value = '';
  toast('✓ Heures enregistrées !');
  renderDashboard();
  renderRecap();
}

/* ════════════════════════════════════════════════════
   DASHBOARD
════════════════════════════════════════════════════ */
function renderDashboard() {
  const ym = currentYM();
  const monthEntries = state.entries.filter(e=>e.date.startsWith(ym));
  const totalH = monthEntries.reduce((s,e)=>s+e.hours, 0);

  document.getElementById('dash-total-h').textContent = totalH;
  document.getElementById('dash-emp-count').textContent = state.employees.length;
  document.getElementById('dash-entry-count').textContent = monthEntries.length;
  document.getElementById('dash-month-label').textContent = ymToLabel(ym);

  const list = document.getElementById('dash-recent-list');
  const recent = [...state.entries].sort((a,b)=>b.date.localeCompare(a.date)).slice(0,8);
  if (!recent.length) {
    list.innerHTML = `<div class="empty"><div class="empty-icon">📋</div><div class="empty-text">Aucune saisie pour l'instant.</div></div>`;
    return;
  }
  list.innerHTML = recent.map(e => {
    const emp = getEmployee(e.empId);
    if (!emp) return '';
    return entryHTML(e, emp);
  }).join('');
}

function entryHTML(e, emp) {
  return `
    <div class="entry-item" style="margin-bottom:8px">
      <div class="entry-info">
        <div class="entry-emp">${esc(emp.name)}</div>
        <div class="entry-meta">${formatDate(e.date)} · ${esc(emp.role)}</div>
        ${e.chantier ? `<span class="entry-chantier">📍 ${esc(e.chantier)}</span>` : ''}
      </div>
      <div class="entry-hours">${e.hours}h</div>
    </div>`;
}

function formatDate(ds) {
  const [y,m,d] = ds.split('-');
  return `${d}/${m}/${y}`;
}

/* ════════════════════════════════════════════════════
   RECAP
════════════════════════════════════════════════════ */
function changeMonth(delta) {
  const [y,m] = state.recapMonth.split('-').map(Number);
  const d = new Date(y, m-1+delta, 1);
  state.recapMonth = `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}`;
  save();
  renderRecap();
}

function renderRecap() {
  const ym = state.recapMonth;
  document.getElementById('recap-month-label').textContent = ymToLabel(ym);
  const monthEntries = state.entries.filter(e=>e.date.startsWith(ym));

  // Summary table
  const tbody = document.getElementById('recap-tbody');
  if (!state.employees.length) {
    tbody.innerHTML = `<tr><td colspan="3" style="text-align:center;color:var(--muted);padding:20px">Aucun salarié</td></tr>`;
  } else {
    let grandTotal = 0;
    const rows = state.employees.map(emp => {
      const h = monthEntries.filter(e=>e.empId===emp.id).reduce((s,e)=>s+e.hours,0);
      grandTotal += h;
      return `<tr>
        <td class="td-name">${esc(emp.name)}</td>
        <td style="color:var(--muted)">${esc(emp.role)}</td>
        <td class="td-hours" style="text-align:right">${h}h</td>
      </tr>`;
    });
    rows.push(`<tr class="row-total">
      <td colspan="2" style="color:var(--muted)">TOTAL</td>
      <td class="td-hours" style="text-align:right;font-size:15px">${grandTotal}h</td>
    </tr>`);
    tbody.innerHTML = rows.join('');
  }

  // Detail list
  const detailEl = document.getElementById('recap-detail-list');
  if (!monthEntries.length) {
    detailEl.innerHTML = `<div class="empty"><div class="empty-icon">📅</div><div class="empty-text">Aucune saisie ce mois-ci.</div></div>`;
    return;
  }
  const sorted = [...monthEntries].sort((a,b)=>b.date.localeCompare(a.date));
  detailEl.innerHTML = sorted.map(e => {
    const emp = getEmployee(e.empId);
    if (!emp) return '';
    return entryHTML(e, emp);
  }).join('');
}

/* ════════════════════════════════════════════════════
   CSV EXPORT
════════════════════════════════════════════════════ */
function exportCSV() {
  const ym = state.recapMonth;
  const entries = state.entries.filter(e=>e.date.startsWith(ym));
  if (!entries.length) { toast('⚠ Aucune donnée à exporter.', 'var(--warn)'); return; }

  const BOM = '\uFEFF'; // pour Excel (accents)
  const header = ['Date','Salarié','Poste','Heures','Chantier'];
  const rows = [...entries]
    .sort((a,b)=>a.date.localeCompare(b.date))
    .map(e => {
      const emp = getEmployee(e.empId) || { name:'?', role:'?' };
      return [formatDate(e.date), emp.name, emp.role, e.hours, e.chantier||''].map(v=>`"${String(v).replace(/"/g,'""')}"`).join(';');
    });

  const csv = BOM + [header.join(';'), ...rows].join('\r\n');
  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `heures_${ym}.csv`;
  a.click();
  URL.revokeObjectURL(url);
  toast('✓ Export CSV téléchargé !');
}

/* ────── UTILS ────── */
function esc(str) {
  return String(str)
    .replace(/&/g,'&amp;')
    .replace(/</g,'&lt;')
    .replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;');
}
